# Load force fields
source leaprc.protein.$forcefield
source leaprc.water.$water_model

# Load protein
COM = loadpdb "$protein_pdb"

# Solvate system
$box_model_solvate COM $water_model_box $box_size

# Add salt and neutralize the system. This is about 0.15M
addIonsRand COM Na+ 0
addIonsRand COM Cl- 0
addIonsRand COM Na+ $n_ions
addIonsRand COM Cl- $n_ions

saveamberparm COM "$working_dir/$name.prmtop" "$working_dir/$name.inpcrd"
quit
